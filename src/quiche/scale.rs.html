27/// Rounding policy for a nonnegative integer multiplied by a floating-point
28/// gain. Nearest breaks ties upward, matching the prior positive `f64::round`.
29#[derive(Clone, Copy)]
30pub enum Rounding {
31    Down,
32    Up,
33    Nearest,
34}
35
36/// Scales a full-width integer by the exact value represented by a binary64
37/// gain, without first rounding the integer to floating point. Negative and
38/// NaN gains produce zero; positive overflow saturates. Zero times infinity
39/// produces zero, as with a saturating float-to-integer conversion.
40///
41/// A finite f64 is an integer significand times a power of two. Multiplying
42/// its at-most-53-bit significand by a u64 needs at most 117 bits. Shifting
43/// that product gives the integer result, with a single final rounding step.
44#[inline]
45pub fn scale_u64(value: u64, gain: f64, rounding: Rounding) -> u64 {
46    u64::try_from(scale_wide(value, gain, rounding)).unwrap_or(u64::MAX)
47}
48
49/// Scale into a wide accumulator, rounding once from the exact binary64
50/// significand product. In particular the integer input never passes through
51/// a floating-point conversion.
52#[inline]
53pub fn scale_wide(value: u64, gain: f64, rounding: Rounding) -> u128 {
54    if value == 0 || gain.is_nan() || gain <= 0.0 {
55        return 0;
56    }
57    if gain.to_bits() == 1.0_f64.to_bits() {
58        return u128::from(value);
59    }
60    if gain.is_infinite() {
61        return u128::MAX;
62    }
63
64    let (product, exponent) = gain_product(value, gain);
65
66    if exponent >= 0 {
67        let shift = exponent.unsigned_abs();
68        if shift > product.leading_zeros() {
69            return u128::MAX;
70        }
71        return product << shift;
72    }
73
74    let shift = exponent.unsigned_abs();
75    if shift >= u128::BITS {
76        // A positive product of at most 117 bits is below half of 2^128.
77        return u128::from(matches!(rounding, Rounding::Up));
78    }
79
80    let quotient = product >> shift;
81    let remainder = product - (quotient << shift);
82    let increment = match rounding {
83        Rounding::Down => false,
84        Rounding::Up => remainder != 0,
85        Rounding::Nearest => remainder >= (1_u128 << (shift - 1)),
86    };
87    quotient + u128::from(increment)
88}

90/// Exact significand product and binary exponent for a positive finite gain.
91#[inline]
92fn gain_product(value: u64, gain: f64) -> (u128, i32) {
93    const FRACTION_BITS: u32 = 52;
94    const EXPONENT_BIAS: i32 = 1023;
95    const SIGNIFICAND_BIT: u64 = 1 << FRACTION_BITS;
96    let bits = gain.to_bits();
97    let encoded_exponent =
98        i32::try_from((bits >> FRACTION_BITS) & 0x7ff).unwrap_or(0);
99    let fraction = bits & (SIGNIFICAND_BIT - 1);
100    let (significand, encoded_exponent) = if encoded_exponent == 0 {
101        (fraction, 1) // subnormals have no implicit leading bit
102    } else {
103        (fraction | SIGNIFICAND_BIT, encoded_exponent)
104    };
105    let product = u128::from(value) * u128::from(significand);
106    (product, encoded_exponent - EXPONENT_BIAS - 52)
107}

109/// Floor an integer times a binary64 gain divided by an integer, saturating
110/// only the final quotient. This avoids both a rounded floating-point divisor
111/// and premature saturation of an intermediate product.
112#[inline]
113pub fn scale_gain_div_u64(value: u64, gain: f64, divisor: u64) -> u64 {
114    if divisor == 0 {
115        return 0;
116    }
117    // Even if the wide numerator saturated, its quotient by any nonzero
118    // u64 divisor must still be at least u64::MAX. Flooring before the
119    // integer division does not change the final floor.
120    u64::try_from(scale_wide(value, gain, Rounding::Down) / u128::from(divisor))
121        .unwrap_or(u64::MAX)
122}

124#[inline]
125pub fn scale_usize(value: usize, gain: f64, rounding: Rounding) -> usize {
126    let value = u64::try_from(value).unwrap_or(u64::MAX);
127    usize::try_from(scale_u64(value, gain, rounding)).unwrap_or(usize::MAX)
128}

130/// Scale by an exact nonnegative rational. Round only once, after the
131/// full-width multiplication; never round the input or the ratio to a float.
132/// A zero denominator represents an unavailable ratio and yields zero.
133#[inline]
134pub fn scale_ratio_u64(
135    value: u64, numerator: u64, denominator: u64, rounding: Rounding,
136) -> u64 {
137    if denominator == 0 {
138        return 0;
139    }
140    let product = u128::from(value) * u128::from(numerator);
141    let denominator = u128::from(denominator);
142    let quotient = product / denominator;
143    let remainder = product % denominator;
144    let increment = match rounding {
145        Rounding::Down => false,
146        Rounding::Up => remainder != 0,
147        Rounding::Nearest => remainder >= denominator.div_ceil(2),
148    };
149    u64::try_from(quotient + u128::from(increment)).unwrap_or(u64::MAX)
150}
151
152#[inline]
153pub fn scale_ratio_usize(
154    value: usize, numerator: u64, denominator: u64, rounding: Rounding,
155) -> usize {
156    usize::try_from(scale_ratio_u64(
157        u64::try_from(value).unwrap_or(u64::MAX),
158        numerator,
159        denominator,
160        rounding,
161    ))
162    .unwrap_or(usize::MAX)
163}

174    #[test]
175    fn wide_scaling_preserves_full_significand_product() {
176        use super::scale_wide;
177
178        for value in [0, 1, 1_u64 << 53, u64::MAX] {
179            assert_eq!(
180                scale_wide(value, 4_294_967_296.0, Rounding::Down),
181                u128::from(value) << 32
182            );
183            assert_eq!(
184                scale_wide(value, 1.5, Rounding::Down),
185                u128::from(value) * 3 / 2
186            );
187        }
188        assert_eq!(
189            scale_wide(1, 2.0_f64.powi(127), Rounding::Down),
190            1_u128 << 127
191        );
192        assert_eq!(scale_wide(2, 2.0_f64.powi(127), Rounding::Down), u128::MAX);
193        assert_eq!(scale_wide(u64::MAX, f64::INFINITY, Rounding::Up), u128::MAX);
194        assert_eq!(scale_wide(1, f64::from_bits(1), Rounding::Up), 1);
195    }

197    #[test]
198    fn gain_division_rounds_once_and_saturates_only_after_division() {
199        for value in [0, 1, 1299, 1300, 1301, (1 << 53) + 1, u64::MAX] {
200            for divisor in [1, 2, 1300, u64::MAX] {
201                for (gain, num, den) in [
202                    (0.25, 1, 4),
203                    (0.5, 1, 2),
204                    (1.0, 1, 1),
205                    (1.5, 3, 2),
206                    (8.0, 8, 1),
207                ] {
208                    let exact =
209                        u128::from(value) * num / (u128::from(divisor) * den);
210                    assert_eq!(
211                        scale_gain_div_u64(value, gain, divisor),
212                        u64::try_from(exact).unwrap_or(u64::MAX)
213                    );
214                }
215            }
216        }
217        assert_eq!(scale_gain_div_u64(u64::MAX, 2.0, 2), u64::MAX);
218        assert_eq!(scale_gain_div_u64(u64::MAX, 2.0, 4), u64::MAX / 2);
219        assert_eq!(scale_gain_div_u64(1, f64::MAX, u64::MAX), u64::MAX);
220        assert_eq!(scale_gain_div_u64(u64::MAX, f64::from_bits(1), 2), 0);
221        assert_eq!(scale_gain_div_u64(123, 1.0, 0), 0);
222        assert_eq!(scale_gain_div_u64(0, f64::INFINITY, 2), 0);
223    }

225    #[test]
226    fn rational_scaling_preserves_integer_loss_and_rounding() {
227        for value in [0, 1, 7, 10, 11, (1 << 53) + 1, u64::MAX] {
228            let product = u128::from(value) * 7;
229            for (rounding, expected) in [
230                (Rounding::Down, product / 10),
231                (Rounding::Up, product.div_ceil(10)),
232                (Rounding::Nearest, (product + 5) / 10),
233            ] {
234                assert_eq!(
235                    scale_ratio_u64(value, 7, 10, rounding),
236                    u64::try_from(expected).unwrap()
237                );
238            }
239            assert_eq!(
240                scale_ratio_u64(value, u64::MAX, u64::MAX, Rounding::Nearest),
241                value
242            );
243        }
244        assert_eq!(scale_ratio_u64(10, 1, 0, Rounding::Nearest), 0);
245        assert_eq!(
246            scale_ratio_u64(u64::MAX, u64::MAX, 1, Rounding::Up),
247            u64::MAX
248        );
249        assert_eq!(
250            scale_ratio_usize(usize::MAX, 7, 10, Rounding::Down),
251            usize::try_from((usize::MAX as u128) * 7 / 10).unwrap()
252        );
253    }

255    #[test]
256    fn exact_integer_scaling_at_precision_boundaries() {
257        let large = (1_u64 << 53) + 1;
258        for (rounding, half) in [
259            (Rounding::Down, large / 2),
260            (Rounding::Up, large.div_ceil(2)),
261            (Rounding::Nearest, large.div_ceil(2)),
262        ] {
263            assert_eq!(scale_u64(large, 1.0, rounding), large);
264            assert_eq!(scale_u64(large, 0.5, rounding), half);
265            assert_eq!(scale_u64(large, 2.0, rounding), large * 2);
266            assert_eq!(scale_u64(u64::MAX, 1.0, rounding), u64::MAX);
267            assert_eq!(scale_u64(u64::MAX, 2.0, rounding), u64::MAX);
268            assert_eq!(scale_u64(0, f64::INFINITY, rounding), 0);
269            assert_eq!(scale_u64(large, f64::INFINITY, rounding), u64::MAX);
270            assert_eq!(scale_u64(large, f64::NAN, rounding), 0);
271            assert_eq!(scale_u64(large, -1.0, rounding), 0);
272        }
273    }

275    #[test]
276    fn fractional_scaling_rounds_once_and_saturates() {
277        let tiny = f64::from_bits(1);
278        assert_eq!(scale_u64(u64::MAX, tiny, Rounding::Up), 1);
279        assert_eq!(scale_u64(u64::MAX, tiny, Rounding::Nearest), 0);
280        assert_eq!(scale_u64(u64::MAX, tiny, Rounding::Down), 0);
281        assert_eq!(scale_u64(u64::MAX, 0.5, Rounding::Down), u64::MAX / 2);
282        assert_eq!(scale_u64(u64::MAX, 0.5, Rounding::Nearest), 1 << 63);
283        assert_eq!(scale_u64(7, 1.25, Rounding::Down), 8);
284        assert_eq!(scale_u64(7, 1.25, Rounding::Up), 9);
285        assert_eq!(scale_u64(7, 1.25, Rounding::Nearest), 9);
286        assert_eq!(scale_u64(1, 2.0_f64.powi(64), Rounding::Down), u64::MAX);
287        assert_eq!(scale_u64(1, 2.0_f64.powi(63), Rounding::Up), 1 << 63);
288        assert_eq!(scale_usize(usize::MAX, 1.0, Rounding::Up), usize::MAX);
289        assert_eq!(scale_usize(usize::MAX, 2.0, Rounding::Down), usize::MAX);
290    }

292    #[test]
293    fn dyadic_gains_match_wide_integer_reference() {
294        // Independently evaluate a set of dyadic factors as rational numbers,
295        // including full-width inputs and values whose f64 conversion loses
296        // bits. No floating-point intermediate is used by the oracle.
297        for value in [0, 1, 7, 65535, 1 << 24, (1 << 53) + 1, u64::MAX] {
298            for numerator in [1_u32, 3, 7, 17, 255, 65535] {
299                for exponent in [0_u32, 1, 4, 16, 31] {
300                    let denominator = 1_u64 << exponent;
301                    let gain = f64::from(numerator) /
302                        2.0_f64.powi(i32::try_from(exponent).unwrap());
303                    let product = u128::from(value) * u128::from(numerator);
304                    let denominator = u128::from(denominator);
305                    for (rounding, expected) in [
306                        (Rounding::Down, product / denominator),
307                        (Rounding::Up, product.div_ceil(denominator)),
308                        (
309                            Rounding::Nearest,
310                            (product + denominator / 2) / denominator,
311                        ),
312                    ] {
313                        assert_eq!(
314                            scale_u64(value, gain, rounding),
315                            u64::try_from(expected).unwrap_or(u64::MAX)
316                        );
317                    }
318                }
319            }
320        }
