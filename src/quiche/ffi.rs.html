133/// Borrow an initialized C object, rejecting null and misaligned pointers.
134///
135/// # Safety
136/// A non-null, aligned pointer must identify a live initialized T for the
137/// returned lifetime. The C caller must not mutate it during that borrow.
138pub unsafe fn ffi_ptr_ref<'a, T>(ptr: *const T) -> Option<&'a T> {
139    if ptr.is_null() || !ptr.is_aligned() {
140        return None;
141    }
142
143    // SAFETY: null and alignment are checked above. Allocation validity and
144    // lifetime remain part of the C caller contract.
145    Some(unsafe { &*ptr })
146}

148/// Exclusively borrow an initialized C object, rejecting null and misalignment.
149///
150/// # Safety
151/// A non-null, aligned pointer must identify a live initialized T, with
152/// exclusive access for the returned lifetime.
153pub unsafe fn ffi_ptr_mut<'a, T>(ptr: *mut T) -> Option<&'a mut T> {
154    if ptr.is_null() || !ptr.is_aligned() {
155        return None;
156    }
157
158    // SAFETY: null and alignment are checked above. Allocation validity,
159    // lifetime and exclusivity remain part of the C caller contract.
160    Some(unsafe { &mut *ptr })
161}

163/// Reclaim an allocation returned by a C constructor once.
164///
165/// # Safety
166/// A non-null, aligned pointer must come from `Box::into_raw` for the same
167/// type and transfer sole ownership to this call.
168pub unsafe fn ffi_drop_box<T>(ptr: *mut T) {
169    if !ptr.is_null() && ptr.is_aligned() {
170        // SAFETY: the caller supplies the original owned box pointer.
171        unsafe { drop(Box::from_raw(ptr)) };
172    }
173}

257/// Borrow writable C output storage without requiring an initialized value.
258///
259/// # Safety
260/// A non-null, aligned pointer must identify writable storage for one T with
261/// exclusive access for the returned lifetime.
262pub unsafe fn ffi_ptr_out<'a, T>(ptr: *mut T) -> Option<&'a mut MaybeUninit<T>> {
263    // SAFETY: MaybeUninit has the same layout as T and permits uninitialized
264    // storage. The caller provides the exclusive writable allocation.
265    unsafe { ffi_ptr_mut(ptr.cast::<MaybeUninit<T>>()) }
266}

285fn ffi_slice_layout_valid<T>(ptr: *const T, len: usize) -> bool {
286    if len == 0 {
287        return true;
288    }
289
290    if ptr.is_null() || !ptr.is_aligned() {
291        return false;
292    }
293
294    size_of::<T>().checked_mul(len).is_some_and(|bytes| {
295        isize::try_from(bytes).is_ok() && ptr.addr().checked_add(bytes).is_some()
296    })
297}

299/// Builds a shared slice from a C pointer after checking the parts that can be
300/// validated without dereferencing caller memory.
301///
302/// # Safety
303///
304/// For non-empty slices, \`ptr\` must still point to \`len\` initialized \`T\`
305/// values in one live allocation for the returned lifetime.
306pub unsafe fn ffi_slice_from_raw_parts<'a, T>(
307    ptr: *const T, len: usize,
308) -> Option<&'a [T]> {
309    if len == 0 {
310        return Some(&[]);
311    }
312
313    if !ffi_slice_layout_valid(ptr, len) {
314        return None;
315    }
316
317    // SAFETY: null/alignment/size arithmetic were checked above. Allocation
318    // validity and lifetime remain part of the C caller contract.
319    Some(unsafe { slice::from_raw_parts(ptr, len) })
320}

322/// Builds a mutable slice from a C pointer after checking the parts that can be
323/// validated without dereferencing caller memory.
324///
325/// # Safety
326///
327/// For non-empty slices, \`ptr\` must point to \`len\` initialized \`T\` values
328/// in one live allocation and be exclusively borrowed for the returned
329/// lifetime.
330pub unsafe fn ffi_slice_from_raw_parts_mut<'a, T>(
331    ptr: *mut T, len: usize,
332) -> Option<&'a mut [T]> {
333    if len == 0 {
334        return Some(&mut []);
335    }
336
337    if !ffi_slice_layout_valid(ptr.cast_const(), len) {
338        return None;
339    }
340
341    // SAFETY: null/alignment/size arithmetic were checked above. Allocation
342    // validity, lifetime and exclusivity remain part of the C caller contract.
343    Some(unsafe { slice::from_raw_parts_mut(ptr, len) })
344}

346/// Borrows a C output buffer without assuming that any bytes are initialized.
347///
348/// # Safety
349/// Non-empty storage must be writable and exclusive for the returned lifetime.
350pub unsafe fn ffi_slice_output<'a, T>(
351    ptr: *mut T, len: usize,
352) -> Option<&'a mut [MaybeUninit<T>]> {
353    // SAFETY: MaybeUninit has T's layout and permits uninitialized storage;
354    // the caller supplies the exclusive output allocation.
355    unsafe { ffi_slice_from_raw_parts_mut(ptr.cast::<MaybeUninit<T>>(), len) }
356}

412/// Borrows a non-null NUL-terminated C string as UTF-8.
413///
414/// # Safety
415///
416/// `value` must point to a valid NUL-terminated C string for the duration of
417/// the returned borrow, without mutation.
418unsafe fn c_str_to_str<'a>(
419    value: *const c_char,
420) -> std::result::Result<&'a str, ()> {
421    if value.is_null() {
422        return Err(());
423    }
424
425    // SAFETY: null was checked above; the C caller owns the valid string.
426    unsafe { ffi::CStr::from_ptr(value) }
427        .to_str()
428        .map_err(|_| ())
429}

437struct Logger {
438    cb: extern "C" fn(line: *const u8, argp: *mut c_void),
439    argp: atomic::AtomicPtr<c_void>,
440}
441
442impl log::Log for Logger {
443    fn enabled(&self, _metadata: &log::Metadata) -> bool {
444        true
445    }
446
447    fn log(&self, record: &log::Record) {
448        let line = format!("{}: {}\0", record.target(), record.args());
449        (self.cb)(line.as_ptr(), self.argp.load(atomic::Ordering::Relaxed));
450    }
451
452    fn flush(&self) {}
453}

455#[no_mangle]
456pub extern "C" fn quiche_enable_debug_logging(
457    cb: Option<extern "C" fn(line: *const u8, argp: *mut c_void)>,
458    argp: *mut c_void,
459) -> c_int {
460    let Some(cb) = cb else {
461        return c_int::from(FFI_ERR_INVALID_ARGUMENT);
462    };
463    let argp = atomic::AtomicPtr::new(argp);
464    let logger = Box::new(Logger { cb, argp });
465
466    if log::set_boxed_logger(logger).is_err() {
467        return -1;
468    }
469
470    log::set_max_level(log::LevelFilter::Trace);
471
472    0
473}

480// The C string configuration entry points have one common validation and
481// conversion path; only the Rust operation differs.
482fn config_string_operation(
483    config: *mut Config, value: *const c_char,
484    op: impl FnOnce(&mut Config, &str) -> Result<()>,
485) -> c_int {
486    let config = ffi_mut!(config, c_int::from(FFI_ERR_INVALID_ARGUMENT));
487    let value = ffi_c_str!(value, c_int::from(FFI_ERR_INVALID_ARGUMENT));
488    op(config, value).map_or_else(|e| c_int::from(e.to_c()), |()| 0)
489}

491// Likewise, the byte-input operations all borrow the validated input for
492// the duration of the call and return a C status code, not a byte count.
493fn bytes_operation<T>(
494    receiver: *mut T, input: *const u8, len: size_t,
495    op: impl FnOnce(&mut T, &[u8]) -> Result<()>,
496) -> c_int {
497    let receiver = ffi_mut!(receiver, c_int::from(FFI_ERR_INVALID_ARGUMENT));
498    let input = ffi_slice!(input, len, c_int::from(FFI_ERR_INVALID_ARGUMENT));
499    op(receiver, input).map_or_else(|e| c_int::from(e.to_c()), |()| 0)
500}

842#[no_mangle]
843pub extern "C" fn quiche_header_info(
844    buf: *const u8, buf_len: size_t, dcil: size_t, version: *mut u32,
845    ty: *mut u8, source_id: *mut u8, source_id_len: *mut size_t,
846    destination_id: *mut u8, destination_id_len: *mut size_t, token: *mut u8,
847    token_len: *mut size_t,
848) -> c_int {
849    let version = ffi_out!(version, c_int::from(FFI_ERR_INVALID_ARGUMENT));
850    let ty = ffi_out!(ty, c_int::from(FFI_ERR_INVALID_ARGUMENT));
851    let source_id_len =
852        ffi_mut!(source_id_len, c_int::from(FFI_ERR_INVALID_ARGUMENT));
853    let destination_id_len =
854        ffi_mut!(destination_id_len, c_int::from(FFI_ERR_INVALID_ARGUMENT));
855    let token_len = ffi_mut!(token_len, c_int::from(FFI_ERR_INVALID_ARGUMENT));
856    let buf = ffi_slice!(buf, buf_len, c_int::from(FFI_ERR_INVALID_ARGUMENT));
857    let hdr = match Header::from_slice(buf, dcil) {
858        Ok(v) => v,
859
860        Err(e) => return c_int::from(e.to_c()),
861    };
862
863    let token_bytes = hdr.token.as_deref().unwrap_or(&[]);
864    if *source_id_len < hdr.scid.len() ||
865        *destination_id_len < hdr.dcid.len() ||
866        *token_len < token_bytes.len()
867    {
868        return c_int::from(Error::BufferTooShort.to_c());
869    }
870    let mut source_id = ffi_slice_out!(
871        source_id,
872        *source_id_len,
873        c_int::from(FFI_ERR_INVALID_ARGUMENT)
874    );
875    let mut destination_id = ffi_slice_out!(
876        destination_id,
877        *destination_id_len,
878        c_int::from(FFI_ERR_INVALID_ARGUMENT)
879    );
880    let mut token = if hdr.token.is_some() {
881        ffi_slice_out!(token, *token_len, c_int::from(FFI_ERR_INVALID_ARGUMENT))
882    } else {
883        &mut []
884    };
885
886    source_id.put_slice(&hdr.scid);
887    destination_id.put_slice(&hdr.dcid);
888    token.put_slice(token_bytes);
889    *source_id_len = hdr.scid.len();
890    *destination_id_len = hdr.dcid.len();
891    *token_len = token_bytes.len();
892    version.write(hdr.version);
893    ty.write(match hdr.ty {
894        Type::Initial => 1,
895        Type::Retry => 2,
896        Type::Handshake => 3,
897        Type::ZeroRTT => 4,
898        Type::Short => 5,
899        Type::VersionNegotiation => 6,
900    });
901
902    0
903}

1287    let conn = ffi_mut!(conn, ssize_t::from(FFI_ERR_INVALID_ARGUMENT));
1288    let info = ffi_ref!(info, ssize_t::from(FFI_ERR_INVALID_ARGUMENT));
1289
1290    if let Err(e) = validate_ssize_len(buf_len) {
1291        return e;
1292    }
1293
1294    // SAFETY: the C receive API requires readable address inputs during the
1295    // call.
1296    let Ok(info) = (unsafe { info.decode() }) else {
1297        return ssize_t::from(FFI_ERR_INVALID_ARGUMENT);
1298    };
1299
1300    let buf =
1301        ffi_slice_mut!(buf, buf_len, ssize_t::from(FFI_ERR_INVALID_ARGUMENT));
1302
1303    match conn.recv(buf, info) {
1304        Ok(v) => ssize_t::try_from(v).unwrap_or(ssize_t::MAX),
1305
1306        Err(e) => ssize_t::from(e.to_c()),
1307    }
1308}

1357    conn: *mut Connection, out: *mut u8, out_len: size_t, from: *const sockaddr,

1369        from,

1379        Ok((v, info)) => {

1381
1382            ssize_t::try_from(v).unwrap_or(ssize_t::MAX)

1391    conn: *mut Connection, stream_id: u64, out: *mut u8, out_len: size_t,
1392    fin: *mut bool, out_error_code: *mut u64,
1393) -> ssize_t {

1400        return e;
1401    }

1404        ffi_slice_out!(out, out_len, ssize_t::from(FFI_ERR_INVALID_ARGUMENT));

1411                Error::StreamReset(error) | Error::StreamStopped(error) => {

1413                },
1414                _ => {},
1415            }
1416            return ssize_t::from(e.to_c());
1417        },
1418    };
1419
1420    fin.write(out_fin);
1421
1422    ssize_t::try_from(out_len).unwrap_or(ssize_t::MAX)
1423}

2504// Each tuple is a value plus its C storage and length output. A path event
2505// writes all its addresses or none if any output pointer is invalid.
2506fn write_path_addresses<const N: usize>(
2507    outputs: [(&SocketAddr, *mut sockaddr_storage, *mut socklen_t); N],
2508) -> bool {
2509    for &(_, address, len) in &outputs {
2510        if address.is_null() ||
2511            !address.is_aligned() ||
2512            len.is_null() ||
2513            !len.is_aligned()
2514        {
2515            return false;
2516        }
2517    }
2518    for (addr, address, len) in outputs {
2519        let (value, size) = std_addr_to_c(addr);
2520        // SAFETY: metadata was validated for every output before any write;
2521        // the C caller supplies writable storage for the declared types.
2522        // No references are formed to caller storage, which may be
2523        // uninitialized.
2524        unsafe {
2525            address.write(value);
2526            len.write(size);
2527        }
2528    }
2529    true
2530}

2696/// Decode a possibly absent C address.
2697///
2698/// # Safety
2699/// A present pointer with an accepted length must identify readable address
2700/// fields for the duration of this call.
2701unsafe fn optional_std_addr_from_c(
2702    addr: *const sockaddr, addr_len: socklen_t,
2703) -> std::result::Result<Option<SocketAddr>, ()> {
2704    match (addr.is_null(), addr_len == 0) {
2705        (true, true) => return Ok(None),
2706        (true, false) | (false, true) => return Err(()),
2707        (false, false) => {},
2708    }
2709
2710    // SAFETY: the caller supplies the same readable address contract.
2711    unsafe { std_addr_from_c(addr, addr_len) }.map(Some)
2712}

2714/// Decode a C address by reading only its initialized address fields.
2715///
2716/// # Safety
2717/// If non-null and its length matches a supported address structure, the
2718/// pointer must refer to readable storage for that structure's address fields.
2719/// No Rust reference is formed to caller memory or its unused padding.
2720unsafe fn std_addr_from_c(
2721    addr: *const sockaddr, addr_len: socklen_t,
2722) -> std::result::Result<SocketAddr, ()> {
2723    let len = usize::try_from(addr_len).map_err(|_| ())?;
2724    if addr.is_null() ||
2725        !addr.is_aligned() ||
2726        (len != size_of::<sockaddr_in>() && len != size_of::<sockaddr_in6>())
2727    {
2728        return Err(());
2729    }
2730
2731    // SAFETY: both accepted layouts contain sa_family at the sockaddr offset;
2732    // the caller guarantees the corresponding field is readable.
2733    let family = unsafe { (&raw const (*addr).sa_family).read() };
2734    // `AF_INET` is a c_int on Unix but an ADDRESS_FAMILY (u16) on Windows.
2735    // Normalize both the incoming family and the platform constants before
2736    // comparing; no lossy narrowing or inferred match-pattern types.
2737    match c_int::from(family) {
2738        found if found == c_int::from(AF_INET) => {
2739            let in4 = addr.cast::<sockaddr_in>();
2740            if len != size_of::<sockaddr_in>() || !in4.is_aligned() {
2741                return Err(());
2742            }
2743
2744            // SAFETY: length, family and alignment match sockaddr_in. Read
2745            // the port and address, not the unused sin_zero bytes.
2746            let (sin_addr, sin_port) = unsafe {
2747                (
2748                    (&raw const (*in4).sin_addr).read(),
2749                    (&raw const (*in4).sin_port).read(),
2750                )
2751            };
2752
2753            #[cfg(not(windows))]
2754            let ip_addr = Ipv4Addr::from(u32::from_be(sin_addr.s_addr));
2755            #[cfg(windows)]
2756            let ip_addr = {
2757                let ip_bytes = unsafe { sin_addr.S_un.S_un_b };
2758
2759                Ipv4Addr::from([
2760                    ip_bytes.s_b1,
2761                    ip_bytes.s_b2,
2762                    ip_bytes.s_b3,
2763                    ip_bytes.s_b4,
2764                ])
2765            };
2766
2767            let port = u16::from_be(sin_port);
2768
2769            let out = SocketAddrV4::new(ip_addr, port);
2770
2771            Ok(out.into())
2772        },
2773
2774        found if found == c_int::from(AF_INET6) => {
2775            let in6 = addr.cast::<sockaddr_in6>();
2776            if len != size_of::<sockaddr_in6>() || !in6.is_aligned() {
2777                return Err(());
2778            }
2779
2780            // SAFETY: length, family and alignment match sockaddr_in6.
2781            let (sin6_addr, sin6_port, flowinfo) = unsafe {
2782                (
2783                    (&raw const (*in6).sin6_addr).read(),
2784                    (&raw const (*in6).sin6_port).read(),
2785                    (&raw const (*in6).sin6_flowinfo).read(),
2786                )
2787            };
2788
2789            let ip_addr = Ipv6Addr::from(
2790                #[cfg(not(windows))]
2791                sin6_addr.s6_addr,
2792                #[cfg(windows)]
2793                unsafe {
2794                    sin6_addr.u.Byte
2795                },
2796            );
2797
2798            let port = u16::from_be(sin6_port);
2799
2800            #[cfg(not(windows))]
2801            // SAFETY: the validated IPv6 layout contains this initialized field.
2802            let scope_id = unsafe { (&raw const (*in6).sin6_scope_id).read() };
2803            #[cfg(windows)]
2804            // SAFETY: read the scope ID from the validated IPv6 union layout.
2805            let scope_id =
2806                unsafe { (&raw const (*in6).Anonymous.sin6_scope_id).read() };
2807
2808            // SocketAddrV6 uses the platform's raw sin6_flowinfo field
2809            // (as std's socket conversion does); preserve it byte-for-byte.
2810            let out = SocketAddrV6::new(ip_addr, port, flowinfo, scope_id);
2811
2812            Ok(out.into())
2813        },
2814
2815        _ => Err(()),
2816    }
2817}

2924#[cfg(not(any(target_os = "macos", target_os = "ios", target_os = "windows")))]
2925fn monotonic_deadline_timespec(
2926    deadline: Instant, anchor: Instant, clock: timespec,
2927) -> Option<timespec> {
2928    // A kernel CLOCK_MONOTONIC reading must be normalized and nonnegative.
2929    // Validate it before using it as the anchor for Rust's opaque Instant.
2930    if clock.tv_sec < 0 || !(0..1_000_000_000).contains(&clock.tv_nsec) {
2931        return None;
2932    }
2933
2934    let base = i128::from(clock.tv_sec)
2935        .checked_mul(1_000_000_000)?
2936        .checked_add(i128::from(clock.tv_nsec))?;
2937    let offset = if deadline >= anchor {
2938        i128::try_from(deadline.duration_since(anchor).as_nanos()).ok()?
2939    } else {
2940        i128::try_from(anchor.duration_since(deadline).as_nanos())
2941            .ok()?
2942            .checked_neg()?
2943    };
2944
2945    let ns = base.checked_add(offset)?;
2946    if ns < 0 {
2947        return None;
2948    }
2949
2950    Some(timespec {
2951        tv_sec: libc::time_t::try_from(ns.div_euclid(1_000_000_000)).ok()?,
2952        tv_nsec: libc::c_long::try_from(ns.rem_euclid(1_000_000_000)).ok()?,
2953    })
2954}

2956#[cfg(not(any(target_os = "macos", target_os = "ios", target_os = "windows")))]
2957fn std_time_to_c(time: &Instant, out: &mut timespec) {
2958    // Bracket the kernel clock reading with Instant samples. Their midpoint
2959    // bounds scheduling skew without depending on Instant's private layout.
2960    let before = Instant::now();
2961    let mut clock = timespec {
2962        tv_sec: 0,
2963        tv_nsec: 0,
2964    };
2965    // SAFETY: clock is initialized, writable, and the kernel writes a
2966    // timespec to it on success.
2967    let sampled = unsafe {
2968        libc::clock_gettime(libc::CLOCK_MONOTONIC, &raw mut clock) == 0
2969    };
2970    let after = Instant::now();
2971
2972    let converted = if sampled {
2973        after
2974            .checked_duration_since(before)
2975            .and_then(|span| before.checked_add(span / 2))
2976            .and_then(|anchor| monotonic_deadline_timespec(*time, anchor, clock))
2977    } else {
2978        None
2979    };
2980
2981    // The C-facing API historically encodes unrepresentable times as zero.
2982    *out = converted.unwrap_or(timespec {
2983        tv_sec: 0,
2984        tv_nsec: 0,
2985    });
2986}
