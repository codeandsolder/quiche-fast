60// RFC 9438 specifies beta = 0.7. Byte counts use the rational 7/10
61// rather than the nearest binary64 representation of that constant.
62fn beta_reduce(bytes: usize) -> usize {
63    crate::scale::scale_ratio_usize(bytes, 7, 10, crate::scale::Rounding::Down)
64}
65
66const C: f64 = 0.4;
67
68/// Threshold for rolling back state, as percentage of lost packets relative to
69/// cwnd.
70const ROLLBACK_THRESHOLD_PERCENT: usize = 20;
71
72/// Minimum threshold for rolling back state, as number of packets.
73const MIN_ROLLBACK_THRESHOLD: usize = 2;

75/// RFC 9438 Reno-friendliness gain before and after reaching `W_max`.
76#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
77enum RenoGain {
78    #[default]
79    Reduced,
80    Full,
81}
82
83impl RenoGain {
84    const fn ratio(self) -> (u128, u128) {
85        match self {
86            Self::Reduced => (9, 17),
87            Self::Full => (1, 1),
88        }
89    }
90}
91
92/// Byte count with 32 fractional bits; a full usize window remains exact.
93/// Floating-point curve evaluation supplies only a displacement from the
94/// exact byte anchor, not an approximation of the entire congestion window.
95#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord)]
96struct Window(u128);
97
98impl Window {
99    const MAX: u128 = (usize::MAX as u128) * Self::ONE;
100    const ONE: u128 = 1 << 32;
101
102    const fn bytes(bytes: usize) -> Self {
103        Self((bytes as u128) * Self::ONE)
104    }
105
106    fn whole(self) -> usize {
107        usize::try_from(self.0 / Self::ONE).unwrap_or(usize::MAX)
108    }
109
110    /// Convert a curve *difference* to floating point for the cube root.
111    /// Each limb is representable exactly; the final sum has the precision
112    /// of binary64, as required by the continuous CUBIC time model.
113    fn continuous(self) -> f64 {
114        let raw = self.0.to_le_bytes();
115        let (words, _) = raw.as_chunks::<4>();
116        let high = f64::from(u32::from_le_bytes(words[2]));
117        let low = f64::from(u32::from_le_bytes(words[1]));
118        let fraction = f64::from(u32::from_le_bytes(words[0]));
119        high.mul_add(4_294_967_296.0, low + fraction / 4_294_967_296.0)
120    }
121
122    fn add_curve(self, mss: usize, coefficient: f64) -> Self {
123        use crate::scale::scale_wide;
124        use crate::scale::Rounding;
125        let magnitude = coefficient.abs() * 4_294_967_296.0;
126        let mss = u64::try_from(mss).unwrap_or(u64::MAX);
127        if coefficient < 0.0 {
128            Self(
129                self.0
130                    .saturating_sub(scale_wide(mss, magnitude, Rounding::Up)),
131            )
132        } else {
133            Self(
134                self.0
135                    .saturating_add(scale_wide(mss, magnitude, Rounding::Down))
136                    .min(Self::MAX),
137            )
138        }
139    }
140
141    fn reno_increment(
142        acked: usize, cwnd: usize, mss: usize, gain: RenoGain,
143    ) -> Self {
144        if cwnd == 0 {
145            return Self::default();
146        }
147        let (numerator, denominator) = gain.ratio();
148        let product = (acked as u128) * (mss as u128);
149        let divisor = cwnd as u128;
150        let factor = Self::ONE * numerator;
151        let scaled = product.checked_mul(factor).map_or_else(
152            || {
153                // Divide before multiplying by the fractional scale when the
154                // intermediate would overflow. Retain the exact remainder.
155                let whole = product / divisor;
156                let remainder = product % divisor;
157                whole
158                    .saturating_mul(factor)
159                    .saturating_add(remainder * factor / divisor)
160            },
161            |product| product / divisor,
162        );
163        Self((scaled / denominator).min(Self::MAX))
164    }
165
166    fn add(self, other: Self) -> Self {
167        Self(self.0.saturating_add(other.0).min(Self::MAX))
168    }
169}

171/// CUBIC State Variables.
172///
173/// We need to keep those variables across the connection.
174/// k, `w_max`, `w_est` are described in the RFC.
175#[derive(Debug, Default)]
176pub struct State {
177    curve: CurveState,
178
179    // Used in CUBIC fix (see on_packet_sent())
180    last_sent_time: Option<Instant>,
181
182    // Tracks when the last ACK was processed, approximating when
183    // bytes_in_flight transitioned toward zero. Used to measure the
184    // actual idle duration in on_packet_sent() instead of the time
185    // since the last send (which inflates the delta by a full RTT).
186    last_ack_time: Option<Instant>,
187
188    // CUBIC state checkpoint preceding the last congestion event.
189    prior: Option<PriorState>,
190}
191
192/// Congestion-avoidance variables that must be restored together after
193/// a spurious congestion event. Send and ACK timestamps are not rolled back.
194#[derive(Clone, Copy, Debug, Default)]
195struct CurveState {
196    // Curve time excludes application-idle intervals. This is not the
197    // packet-send-time cutoff of the latest congestion event.
198    epoch_start: Option<Instant>,
199    k: f64,
200    w_max: Window,
201    w_est: Window,
202    reno_gain: RenoGain,
203    cwnd_inc: usize,
204}
205
206/// Stores the CUBIC state from before the last congestion event.
207/// <https://tools.ietf.org/id/draft-ietf-tcpm-rfc8312bis-00.html#section-4.9>
208#[derive(Debug)]
209struct PriorState {
210    congestion_window: usize,
211    ssthresh: usize,
212    curve: CurveState,
213    recovery_start: Option<Instant>,
214    lost_count: usize,
215}

217/// CUBIC Functions.
218///
219/// Note that these calculations are based on a count of cwnd as bytes,
220/// not packets.
221/// Unit of t (duration) and RTT are based on seconds (f64).
222impl CurveState {
223    // K = cubic_root ((W_max - cwnd) / MSS / C) (Eq. 2).
224    fn cubic_k(&self, cwnd: usize, max_datagram_size: usize) -> f64 {
225        let difference =
226            Window(self.w_max.0.saturating_sub(Window::bytes(cwnd).0));
227        libm::cbrt(
228            difference.continuous() /
229                (C * Window::bytes(max_datagram_size).continuous()),
230        )
231    }
232
233    // W_cubic(t) = W_max + MSS * C * (t - K)^3 (Eq. 1).
234    fn w_cubic(&self, t: Duration, max_datagram_size: usize) -> Window {
235        self.w_max
236            .add_curve(max_datagram_size, C * (t.as_secs_f64() - self.k).powi(3))
237    }
238
239    fn update_reno_estimate(&mut self, acked: usize, cwnd: usize, mss: usize) {
240        self.w_est = self.w_est.add(Window::reno_increment(
241            acked,
242            cwnd,
243            mss,
244            self.reno_gain,
245        ));
246        if self.w_est >= self.w_max {
247            self.reno_gain = RenoGain::Full;
248        }
249    }
250}

254fn on_packet_sent(
255    r: &mut Congestion, sent_bytes: usize, bytes_in_flight: usize, now: Instant,
256) {
257    reno::on_packet_sent(r, sent_bytes, bytes_in_flight, now);
258
259    // Non-data packets must not affect the idle estimate.
260    if sent_bytes == 0 {
261        return;
262    }
263
264    // RFC 9438: suspend curve growth while application-limited. Only
265    // the curve epoch moves; the recovery cutoff is the actual time of
266    // the congestion event and must not move with it.
267    let cubic = &mut r.cubic_state;
268    if bytes_in_flight == 0 {
269        if let (Some(epoch), Some(idle_start)) = (
270            cubic.curve.epoch_start,
271            cmp::max(cubic.last_ack_time, cubic.last_sent_time),
272        ) {
273            let idle = now.saturating_duration_since(idle_start.max(epoch));
274            // idle <= now - epoch, so the adjusted epoch cannot exceed now.
275            cubic.curve.epoch_start = Some(epoch + idle);
276        }
277    }
278
279    cubic.last_sent_time = Some(now);
280}

291fn on_packet_acked(
292    r: &mut Congestion, bytes_in_flight: usize, packet: &Acked, now: Instant,
293    rtt_stats: &RttStats,
294) {
295    if !packet.in_flight || packet.size == 0 {
296        return;
297    }
298    r.cubic_state.last_ack_time = Some(now);
299    let in_congestion_recovery = r.in_congestion_recovery(packet.time_sent);
300
301    if in_congestion_recovery {
302        r.prr.on_packet_acked(
303            packet.size,
304            bytes_in_flight,
305            r.ssthresh.get(),
306            r.max_datagram_size,
307        );
308
309        return;
310    }
311
312    if r.app_limited {
313        return;
314    }
315
316    // Detecting spurious congestion events.
317    // <https://tools.ietf.org/id/draft-ietf-tcpm-rfc8312bis-00.html#section-4.9>
318    //
319    // If recovery ends after fewer than
320    // `cwnd / mss * ROLLBACK_THRESHOLD_PERCENT` packets, treat the congestion
321    // event as spurious and restore the previous state.
322    if let Some(prior) = &r.cubic_state.prior {
323        let new_lost = r.lost_count.saturating_sub(prior.lost_count);
324
325        let rollback_threshold = crate::scale::scale_ratio_usize(
326            r.congestion_window / r.max_datagram_size,
327            ROLLBACK_THRESHOLD_PERCENT as u64,
328            100,
329            crate::scale::Rounding::Down,
330        );
331
332        let rollback_threshold = rollback_threshold.max(MIN_ROLLBACK_THRESHOLD);
333
334        if r.congestion_recovery_start_time.is_some() &&
335            new_lost < rollback_threshold
336        {
337            let did_rollback = rollback(r);
338            if did_rollback {
339                return;
340            }
341        }
342    }
343
344    if r.congestion_window < r.ssthresh.get() {
345        slow_start_ack(r, packet, now, rtt_stats);
346    } else {
347        congestion_avoidance_ack(r, packet, now, rtt_stats);
348    }
349}

351fn slow_start_ack(
352    r: &mut Congestion, packet: &Acked, now: Instant, rtt_stats: &RttStats,
353) {
354    // In Slow start, bytes_acked_sl is used for counting
355    // acknowledged bytes.
356    let acked = r.bytes_acked_sl as u128 + packet.size as u128;
357    let remainder = if acked >= r.max_datagram_size as u128 {
358        let increase = if r.hystart.in_css() {
359            r.hystart.css_cwnd_inc(r.max_datagram_size)
360        } else {
361            r.max_datagram_size
362        };
363        r.congestion_window = r.congestion_window.saturating_add(increase);
364
365        acked - r.max_datagram_size as u128
366    } else {
367        acked
368    };
369    r.bytes_acked_sl = usize::try_from(remainder).unwrap_or(usize::MAX);
370
371    if r.hystart.on_packet_acked(packet, rtt_stats.latest_rtt, now) {
372        // Exit to congestion avoidance if CSS ends.
373        r.ssthresh.update(r.congestion_window, true);
374    }
375}

377fn congestion_avoidance_ack(
378    r: &mut Congestion, packet: &Acked, now: Instant, rtt_stats: &RttStats,
379) {
380    let curve = &mut r.cubic_state.curve;
381    let ca_start_time = *curve.epoch_start.get_or_insert_with(|| {
382        curve.w_max = Window::bytes(r.congestion_window);
383        curve.k = 0.0;
384        curve.w_est = Window::bytes(r.congestion_window);
385        curve.reno_gain = RenoGain::Reduced;
386        r.hystart.css_start_time().unwrap_or(now)
387    });
388
389    let t = now.saturating_duration_since(ca_start_time);
390
391    // target = w_cubic(t + rtt)
392    let target = r
393        .cubic_state
394        .curve
395        .w_cubic(t.saturating_add(*rtt_stats.min_rtt), r.max_datagram_size);
396
397    // Clip to [cwnd, 1.5 * cwnd] in the exact integer byte domain.
398    let target = target.whole().clamp(
399        r.congestion_window,
400        r.congestion_window.saturating_add(r.congestion_window / 2),
401    );
402
403    r.cubic_state.curve.update_reno_estimate(
404        packet.size,
405        r.congestion_window,
406        r.max_datagram_size,
407    );
408
409    let mut cubic_cwnd = r.congestion_window;
410
411    if r.cubic_state.curve.w_cubic(t, r.max_datagram_size) <
412        r.cubic_state.curve.w_est
413    {
414        // AIMD friendly region (W_cubic(t) < W_est)
415        cubic_cwnd = cmp::max(cubic_cwnd, r.cubic_state.curve.w_est.whole());
416    } else {
417        // Concave region or convex region use same increment.
418        let delta = target.saturating_sub(cubic_cwnd);
419        // The product may exceed the address width before division,
420        // especially for multi-megabyte congestion windows on 32-bit.
421        let cubic_inc = if cubic_cwnd == 0 {
422            0
423        } else if let Some(product) = r.max_datagram_size.checked_mul(delta) {
424            product / cubic_cwnd
425        } else {
426            usize::try_from(
427                r.max_datagram_size as u128 * delta as u128 / cubic_cwnd as u128,
428            )
429            .unwrap_or(usize::MAX)
430        };
431
432        cubic_cwnd = cubic_cwnd.saturating_add(cubic_inc);
433    }
434
435    // Update the increment and increase cwnd by MSS.
436    let increment = r.cubic_state.curve.cwnd_inc as u128 +
437        (cubic_cwnd - r.congestion_window) as u128;
438    let remainder = if increment >= r.max_datagram_size as u128 {
439        r.congestion_window =
440            r.congestion_window.saturating_add(r.max_datagram_size);
441        increment - r.max_datagram_size as u128
442    } else {
443        increment
444    };
445    r.cubic_state.curve.cwnd_inc =
446        usize::try_from(remainder).unwrap_or(usize::MAX);
447}

449fn congestion_event(
450    r: &mut Congestion, bytes_in_flight: usize, _lost_bytes: usize,
451    largest_lost_pkt: &Sent, now: Instant,
452) {
453    let time_sent = largest_lost_pkt.time_sent;
454    let in_congestion_recovery = r.in_congestion_recovery(time_sent);
455
456    // Start a new congestion event if packet was sent after the
457    // start of the previous congestion recovery period.
458    if !in_congestion_recovery {
459        r.congestion_recovery_start_time = Some(now);
460        r.cubic_state.curve.epoch_start = Some(now);
461
462        // Fast convergence uses the exact rational (1 + beta) / 2 = 17/20.
463        let window = Window::bytes(r.congestion_window);
464        r.cubic_state.curve.w_max = if window < r.cubic_state.curve.w_max {
465            Window(window.0 * 17 / 20)
466        } else {
467            window
468        };
469
470        let ssthresh = beta_reduce(r.congestion_window);
471        let ssthresh = cmp::max(
472            ssthresh,
473            r.max_datagram_size.saturating_mul(MINIMUM_WINDOW_PACKETS),
474        );
475        r.ssthresh.update(ssthresh, r.hystart.in_css());
476        r.congestion_window = ssthresh;
477
478        r.cubic_state.curve.k =
479            if r.cubic_state.curve.w_max < Window::bytes(r.congestion_window) {
480                0.0
481            } else {
482                r.cubic_state
483                    .curve
484                    .cubic_k(r.congestion_window, r.max_datagram_size)
485            };
486
487        r.cubic_state.curve.cwnd_inc = beta_reduce(r.cubic_state.curve.cwnd_inc);
488
489        r.cubic_state.curve.w_est = Window::bytes(r.congestion_window);
490        r.cubic_state.curve.reno_gain = RenoGain::Reduced;
491
492        if r.hystart.in_css() {
493            r.hystart.congestion_event();
494        }
495
496        r.prr.congestion_event(bytes_in_flight);
497    }
498}

500const fn checkpoint(r: &mut Congestion) {
501    r.cubic_state.prior = Some(PriorState {
502        congestion_window: r.congestion_window,
503        ssthresh: r.ssthresh.get(),
504        curve: r.cubic_state.curve,
505        recovery_start: r.congestion_recovery_start_time,
506        lost_count: r.lost_count,
507    });
508}
509
510fn rollback(r: &mut Congestion) -> bool {
511    let Some(prior) = &r.cubic_state.prior else {
512        return false;
513    };
514
515    // Don't go back to slow start or decrease the window.
516    if prior.congestion_window < prior.ssthresh ||
517        r.congestion_window >= prior.congestion_window
518    {
519        return false;
520    }
521
522    r.congestion_window = prior.congestion_window;
523    r.ssthresh.update(prior.ssthresh, false);
524    r.cubic_state.curve = prior.curve;
525    r.congestion_recovery_start_time = prior.recovery_start;
526    r.prr = super::prr::PRR::default();
527    r.cubic_state.prior = None;
528
529    true
530}

1108    fn cubic_perpetual_recovery_trap() {
1109        // Reproduces the bug where cwnd stays pinned at minimum after a
1110        // loss event because the idle-time epoch shift pushes
1111        // congestion_recovery_start_time into the future on every
1112        // send -> ACK -> send cycle when bif transiently hits 0.
1113        let mut sender = test_sender();
1114        let size = sender.max_datagram_size;
1115        let rtt = Duration::from_millis(100);
1116
1117        // Fill the pipe.
1118        for _ in 0..sender.initial_congestion_window_packets {
1119            sender.send_packet(size);
1120        }
1121
1122        sender.update_rtt(rtt);
1123        sender.advance_time(rtt);
1124
1125        // Trigger a loss to enter recovery and reduce cwnd.
1126        let initial_cwnd = sender.congestion_window;
1127        sender.lose_n_packets(1, size, None);
1128        let post_loss_cwnd = sender.congestion_window;
1129        assert_eq!(post_loss_cwnd, beta_reduce(initial_cwnd));
1130        let initial_recovery_start_time =
1131            sender.congestion_recovery_start_time.unwrap();
1132        assert_eq!(initial_recovery_start_time, sender.time);
1133
1134        // ACK remaining in-flight packets to exit recovery.
1135        sender.ack_n_packets(sender.initial_congestion_window_packets - 1, size);

1139        // minimum cwnd, ACK it (bif drops to 0), advance one RTT, repeat.
1140        // With the bug, recovery_start_time would advance on every

1142        let packets_per_burst = cmp::max(1, sender.congestion_window / size);

1145            for _ in 0..packets_per_burst {
1146                sender.send_packet(size);
1147            }
1148
1149            sender.advance_time(rtt);

1151            assert_eq!(sender.bytes_in_flight, 0);
1152        }

1154        // cwnd must have grown beyond the post-loss value.
1155        assert!(
1156            sender.congestion_window > post_loss_cwnd,
1157            "cwnd stuck at {} (post-loss {}): perpetual recovery trap",
1158            sender.congestion_window,
1159            post_loss_cwnd,
1160        );
1161        // Recovery start hasn't changed.
1162        assert_eq!(
1163            sender.congestion_recovery_start_time.unwrap() -

1238    fn cubic_zero_bytes_sent_no_epoch_shift() {
1239        // A zero-byte packet (e.g. PADDING-only) must NOT shift
1240        // congestion_recovery_start_time or update last_sent_time.
1241        let mut sender = test_sender();
1242        let size = sender.max_datagram_size;
1243        let rtt = Duration::from_millis(100);
1244
1245        // Fill pipe, trigger loss to set congestion_recovery_start_time.
1246        for _ in 0..sender.initial_congestion_window_packets {
1247            sender.send_packet(size);
1248        }
1249        sender.update_rtt(rtt);
1250        sender.advance_time(rtt);
1251        sender.lose_n_packets(1, size, None);
1252
1253        // ACK everything → bif = 0.
1254        sender.ack_n_packets(sender.initial_congestion_window_packets - 1, size);
1255        assert_eq!(sender.bytes_in_flight, 0);
1256
1257        // Record state before zero-byte send.
1258        let recovery_start_before = sender.congestion_recovery_start_time;
1259        let curve_epoch_before = sender.cubic_state.curve.epoch_start;
1260        let last_sent_before = sender.cubic_state.last_sent_time;
1261
1262        // Advance time so any shift would be visible.
1263        sender.advance_time(Duration::from_millis(500));
1264
1265        // Send a zero-byte packet (simulates PADDING-only).
1266        sender.send_packet(0);
1267
1268        // Neither recovery_start_time nor last_sent_time should change.
1269        assert_eq!(
1270            sender.congestion_recovery_start_time, recovery_start_before,
1271            "recovery_start_time must not shift on zero-byte send",
1272        );
1273        assert_eq!(sender.cubic_state.curve.epoch_start, curve_epoch_before);
1274        assert_eq!(
1275            sender.cubic_state.last_sent_time, last_sent_before,
1276            "last_sent_time must not update on zero-byte send",
1277        );
