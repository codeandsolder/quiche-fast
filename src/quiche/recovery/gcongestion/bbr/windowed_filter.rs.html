5// Implements Kathleen Nichols' algorithm for tracking the minimum (or maximum)
6// estimate of a stream of samples over some fixed time interval. (E.g.,
7// the minimum RTT over the past five minutes.) The algorithm keeps track of
8// the best, second best, and third best min (or max) estimates, maintaining an
9// invariant that the measurement time of the n'th best >= n-1'th best.
10
11// The algorithm works as follows. On a reset, all three estimates are set to
12// the same sample. The second best estimate is then recorded in the second
13// quarter of the window, and a third best estimate is recorded in the second
14// half of the window, bounding the worst case error when the true min is
15// monotonically increasing (or true max is monotonically decreasing) over the
16// window.
17//
18// A new best sample replaces all three estimates, since the new best is lower
19// (or higher) than everything else in the window and it is the most recent.
20// The window thus effectively gets reset on every new min. The same property
21// holds true for second best and third best estimates. Specifically, when a
22// sample arrives that is better than the second best but not better than the
23// best, it replaces the second and third best estimates but not the best
24// estimate. Similarly, a sample that is better than the third best estimate
25// but not the other estimates replaces only the third best estimate.
26//
27// Finally, when the best expires, it is replaced by the second best, which in
28// turn is replaced by the third best. The newest sample replaces the third
29// best.
30
31use std::ops::Div;
32use std::ops::Sub;
33
34#[derive(Debug, Clone, Copy)]
35struct Sample<T, I> {
36    sample: T,
37    time: I,
38}
39
40#[derive(Debug)]
41pub struct WindowedFilter<T, I, D> {
42    window_length: D,
43    estimates: Option<[Sample<T, I>; 3]>,
44}
45
46impl<T, I, D> WindowedFilter<T, I, D>
47where
48    T: Ord + Copy,
49    I: Sub<I, Output = D> + Copy,
50    D: Ord + Div<usize, Output = D> + Copy,
51{
52    pub const fn new(window_length: D) -> Self {
53        Self {
54            window_length,
55            estimates: None,
56        }
57    }
58
59    pub fn get_best(&self) -> Option<T> {
60        self.estimates.map(|[best, _, _]| best.sample)
61    }
62
63    /// Recompute stored values in measurement order, retaining their times.
64    /// Returning None discards a sample.
65    pub fn recompute(&mut self, mut update: impl FnMut(T) -> Option<T>) {
66        if let Some(estimates) = self.estimates.take() {
67            for estimate in estimates {
68                if let Some(sample) = update(estimate.sample) {
69                    self.update(sample, estimate.time);
70                }
71            }
72        }
73    }
74
75    pub fn update(&mut self, new_sample: T, new_time: I) {
76        let new = Sample {
77            sample: new_sample,
78            time: new_time,
79        };
80
81        let Some(estimates) = &mut self.estimates else {
82            self.estimates = Some([new; 3]);
83            return;
84        };
85        let [best, second, third] = estimates;
86
87        // Reset for a new best or when even the newest estimate is too old.
88        if new_sample > best.sample || new_time - third.time > self.window_length
89        {
90            *estimates = [new; 3];
91            return;
92        }
93
94        if new_sample > second.sample {
95            *second = new;
96            *third = new;
97        } else if new_sample > third.sample {
98            *third = new;
99        }
100
101        if new_time - best.time > self.window_length {
102            *best = *second;
103            *second = *third;
104            *third = new;
105            if new_time - best.time > self.window_length {
106                *best = *second;
107                *second = *third;
108            }
109            return;
110        }
111
112        // Take a new second estimate in the second quarter of the window.
113        if second.sample == best.sample &&
114            new_time - second.time > self.window_length / 4
115        {
116            *second = new;
117            *third = new;
118            return;
119        }
120
121        // Take a new third estimate in the second half of the window.
122        if third.sample == second.sample &&
123            new_time - third.time > self.window_length / 2
124        {
125            *third = new;
126        }
127    }
128}
