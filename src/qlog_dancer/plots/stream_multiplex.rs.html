95/// Determine stream visibility using the protocol's stream-ID rules. HTTP/2
96/// stream IDs have no QUIC initiator/direction bits.
97fn visible_stream(
98    params: &MultiplexPlotsParams, protocol: crate::datastore::ApplicationProto,
99    stream_id: u64,
100) -> bool {
101    !params.bidi_only ||
102        protocol == crate::datastore::ApplicationProto::Http2 ||
103        (stream_id & 2) == 0
104}

106/// An explicit, non-overlapping vertical layout. Row spacing follows the
107/// configured swimlane heights, rather than fixed 20/70px increments that
108/// overlapped adjacent charts in the original implementation.
109#[derive(Debug, Clone, PartialEq, Eq)]
110struct MultiplexLayout {
111    width: u32,
112    height: u32,
113    chart_width: u32,
114    data_mini_y: u32,
115    bubbles_y: u32,
116    stream_combined_y: u32,
117    stream_mini_y: u32,
118    frame_step: u32,
119    bubble_step: u32,
120}
121
122impl MultiplexLayout {
123    fn new(
124        params: &MultiplexPlotsParams, request_count: usize,
125    ) -> io::Result<Self> {
126        let chart_width = params
127            .width
128            .checked_sub(params.right_shrink_margin)
129            .filter(|width| *width > 0)
130            .ok_or_else(|| {
131                invalid_chart("multiplex width must exceed its right margin")
132            })?;
133        if params.combined_plot_height == 0 ||
134            params.frame_swimlane_height == 0 ||
135            params.bubbles_swimlane_height == 0
136        {
137            return Err(invalid_chart(
138                "multiplex section heights must be nonzero",
139            ));
140        }
141        let rows = u64::try_from(request_count)
142            .map_err(|_| invalid_chart("too many multiplex request rows"))?;
143        let overflow = || invalid_chart("multiplex layout arithmetic overflow");
144        let add = |a: u64, b: u64| a.checked_add(b).ok_or_else(overflow);
145        let mul = |a: u64, b: u64| a.checked_mul(b).ok_or_else(overflow);
146        let spacing = u64::from(params.y_spacer);
147        let frame_step = add(u64::from(params.frame_swimlane_height), spacing)?;
148        let bubble_step =
149            add(u64::from(params.bubbles_swimlane_height), spacing)?;
150        let frame_rows_height = mul(rows, frame_step)?;
151        let bubble_rows_height = mul(rows, bubble_step)?;
152        let data_mini_y = add(u64::from(params.combined_plot_height), spacing)?;
153        let bubbles_y = add(add(data_mini_y, frame_rows_height)?, spacing)?;
154        let stream_combined_y =
155            add(add(bubbles_y, bubble_rows_height)?, spacing)?;
156        let stream_mini_y = add(
157            add(stream_combined_y, u64::from(params.combined_plot_height))?,
158            spacing,
159        )?;
160        let full_height =
161            add(add(add(stream_mini_y, frame_rows_height)?, spacing)?, 40)?;
162        checked_pixel(u64::from(params.width))?;
163        checked_pixel(full_height)?;
164        for position in [data_mini_y, bubbles_y, stream_combined_y, stream_mini_y]
165        {
166            checked_pixel(position)?;
167        }
168        let convert = |v: u64| {
169            u32::try_from(v).map_err(|_| {
170                invalid_chart("multiplex plot coordinate exceeds u32")
171            })
172        };
173        Ok(Self {
174            width: params.width,
175            height: convert(full_height)?,
176            chart_width,
177            data_mini_y: convert(data_mini_y)?,
178            bubbles_y: convert(bubbles_y)?,
179            stream_combined_y: convert(stream_combined_y)?,
180            stream_mini_y: convert(stream_mini_y)?,
181            frame_step: convert(frame_step)?,
182            bubble_step: convert(bubble_step)?,
183        })
184    }
185
186    fn row_y(offset: u32, step: u32, row: usize) -> io::Result<i32> {
187        let row = u64::try_from(row)
188            .map_err(|_| invalid_chart("row index exceeds u64"))?;
189        let y = row
190            .checked_mul(u64::from(step))
191            .and_then(|offset_y| u64::from(offset).checked_add(offset_y))
192            .ok_or_else(|| {
193                invalid_chart("multiplex request row coordinate overflow")
194            })?;
195        checked_pixel(y)
196    }
197}

199/// Select the recorded DATA frames for the trace vantage without collecting a
200/// second vector of per-request references.
201fn request_data_frames(
202    request: &HttpRequestStub, vantage: VantagePoint,
203) -> &[(f64, u64)] {
204    match vantage {
205        VantagePoint::Client => &request.time_data_rx_set,
206        VantagePoint::Server => &request.time_data_tx_set,
207    }
208}

210/// Assign one deterministic color to every visible stream ID represented by
211/// either the request list or the STREAM/DATA frame series. Some streams appear
212/// in only one source; independent palette iteration would shift the colors
213/// of later streams between combined and per-request panels.
214fn colors_for_visible_streams(
215    params: &MultiplexPlotsParams, ds: &Datastore,
216    stream_frame_series: &BTreeMap<u64, Vec<QlogPointu64>>,
217) -> BTreeMap<u64, RGBColor> {
218    let ids: BTreeSet<u64> = ds
219        .http_requests
220        .keys()
221        .chain(stream_frame_series.keys())
222        .copied()
223        .filter(|&id| visible_stream(params, ds.application_proto, id))
224        .collect();
225
226    let mut palette = ColorCycle::default();
227    ids.into_iter()
228        .map(|id| (id, palette.next_color()))
229        .collect()
230}

232/// Draw per-stream DATA and STREAM events across combined and individual lanes.
233///
234/// # Errors
235/// Returns invalid layout, nonfinite times, output filesystem/canvas, and
236/// backend/drawing failures as errors rather than panicking.
237pub fn plot_stream_multiplexing(
238    params: &MultiplexPlotsParams, filename: &str, ss: &SeriesStore,
239    ds: &Datastore, ty: &ChartOutputType,
240) -> Result<(), Box<dyn Error>> {
241    let requests = ds.http_requests.iter().filter(|(stream_id, _)| {
242        visible_stream(params, ds.application_proto, **stream_id)
243    });
244    let layout = MultiplexLayout::new(params, requests.clone().count())?;
245
246    let stream_frame_series = match (ds.application_proto, ds.vantage_point) {
247        (crate::datastore::ApplicationProto::Http2, VantagePoint::Client) =>
248            &ss.received_data_frames_series,
249        (crate::datastore::ApplicationProto::Http2, VantagePoint::Server) =>
250            &ss.sent_data_frames_series,
251        (crate::datastore::ApplicationProto::Http3, VantagePoint::Client) =>
252            &ss.received_stream_frames_series,
253        (crate::datastore::ApplicationProto::Http3, VantagePoint::Server) =>
254            &ss.sent_stream_frames_series,
255    };
256    let (mut start, mut end) = match ds.vantage_point {
257        VantagePoint::Client => (ss.received_x_min, ss.received_x_max),
258        VantagePoint::Server => (ss.sent_x_min, ss.sent_x_max),
259    };
260    if !start.is_finite() || !end.is_finite() {
261        return Err(
262            invalid_chart("multiplex source time bounds must be finite").into()
263        );
264    }
265    for time in requests
266        .clone()
267        .flat_map(|(_, request)| {
268            request_data_frames(request, ds.vantage_point)
269                .iter()
270                .map(|point| point.0)
271        })
272        .chain(
273            stream_frame_series
274                .iter()
275                .filter(|(id, _)| {
276                    visible_stream(params, ds.application_proto, **id)
277                })
278                .flat_map(|(_, points)| points.iter().map(|point| point.0)),
279        )
280        .chain(
281            requests
282                .clone()
283                .flat_map(|(_, request)| {
284                    [request.time_first_headers_rx, request.time_first_headers_tx]
285                })
286                .flatten(),
287        )
288    {
289        if !time.is_finite() {
290            return Err(invalid_chart(
291                "multiplex event timestamps must be finite",
292            )
293            .into());
294        }
295        start = start.min(time);
296        end = end.max(time);
297    }
298    let x_range = checked_time_range(start, end, &params.clamp)?;
299
300    let config =
301        ChartConfig::new("stream-multiplexing", filename, &params.clamp, ds, ty);
302    let margin = ChartMargin {
303        top: 20,
304        bottom: 20,
305        left: 20,
306        right: 20,
307    };
308    #[cfg(not(target_arch = "wasm32"))]
309    let chart_path = match ty {
310        ChartOutputType::Png { .. } => config.chart_filepath(),
311        ChartOutputType::Canvas { .. } =>
312            return Err(invalid_chart(
313                "native multiplex chart requires PNG output",
314            )
315            .into()),
316    };
317    #[cfg(not(target_arch = "wasm32"))]
318    let root = bitmap_area(
319        &chart_path,
320        ChartSize {
321            width: layout.width,
322            height: layout.height,
323        },
324        params.colors,
325        margin,
326    )?;
327
328    #[cfg(target_arch = "wasm32")]
329    let canvas_id = config.canvas_id().ok_or_else(|| {
330        invalid_chart("browser multiplex chart requires canvas output")
331    })?;
332    #[cfg(target_arch = "wasm32")]
333    let root = canvas_area(&canvas_id, params.colors, margin)?;
334
335    let stream_colors =
336        colors_for_visible_streams(params, ds, stream_frame_series);
337    let combined_data_area = root
338        .clone()
339        .shrink((0, 0), (layout.chart_width, params.combined_plot_height));
340    let mut combined_data = ChartBuilder::on(&combined_data_area)
341        .caption(
342            "DATA frame events",
343            chart_title_style(&params.colors.caption),
344        )
345        .x_label_area_size(params.area_margin.x)
346        .y_label_area_size(params.area_margin.y)
347        .build_cartesian_2d(x_range.clone(), 0..2)?;
348    combined_data
349        .configure_mesh()
350        .disable_mesh()
351        .axis_style(params.colors.axis)
352        .label_style(chart_label_style(&params.colors.caption))
353        .draw()?;
354    for (stream_id, request) in requests.clone() {
355        let points = request_data_frames(request, ds.vantage_point);
356        let color = stream_colors[stream_id];
357        combined_data.draw_series(points.iter().map(|(time, _)| {
358            PathElement::new([(*time, 2), (*time, 0)], color)
359        }))?;
360    }
361
362    let combined_stream_area = root.clone().shrink(
363        (0, checked_pixel(u64::from(layout.stream_combined_y))?),
364        (layout.chart_width, params.combined_plot_height),
365    );
366    let mut combined_stream = ChartBuilder::on(&combined_stream_area)
367        .caption(
368            "STREAM frame events",
369            chart_title_style(&params.colors.caption),
370        )
371        .x_label_area_size(params.area_margin.x)
372        .y_label_area_size(params.area_margin.y)
373        .build_cartesian_2d(x_range.clone(), 0..2)?;
374    combined_stream
375        .configure_mesh()
376        .disable_mesh()
377        .axis_style(params.colors.axis)
378        .label_style(chart_label_style(&params.colors.caption))
379        .draw()?;
380    for (stream_id, points) in stream_frame_series
381        .iter()
382        .filter(|(id, _)| visible_stream(params, ds.application_proto, **id))
383    {
384        let color = stream_colors[stream_id];
385        combined_stream.draw_series(points.iter().map(|(time, _)| {
386            PathElement::new([(*time, 2), (*time, 0)], color)
387        }))?;
388    }
389
390    let max_data = requests
391        .clone()
392        .flat_map(|(_, request)| {
393            request_data_frames(request, ds.vantage_point)
394                .iter()
395                .map(|point| point.1)
396        })
397        .max()
398        .unwrap_or(0)
399        .max(1);
400    for (row, (stream_id, request)) in requests.enumerate() {
401        let points = request_data_frames(request, ds.vantage_point);
402        let color = stream_colors[stream_id];
403        let frame_y =
404            MultiplexLayout::row_y(layout.data_mini_y, layout.frame_step, row)?;
405        let stream_y =
406            MultiplexLayout::row_y(layout.stream_mini_y, layout.frame_step, row)?;
407        let bubble_y =
408            MultiplexLayout::row_y(layout.bubbles_y, layout.bubble_step, row)?;
409        let data_area = root.clone().shrink(
410            (0, frame_y),
411            (layout.chart_width, params.frame_swimlane_height),
412        );
413        let stream_area = root.clone().shrink(
414            (0, stream_y),
415            (layout.chart_width, params.frame_swimlane_height),
416        );
417        let bubble_area = root.clone().shrink(
418            (0, bubble_y),
419            (layout.chart_width, params.bubbles_swimlane_height),
420        );
421
422        let mut data_chart = ChartBuilder::on(&data_area)
423            .x_label_area_size(params.area_margin.x)
424            .y_label_area_size(params.area_margin.y)
425            .build_cartesian_2d(x_range.clone(), 0..4)?;
426        data_chart
427            .configure_mesh()
428            .disable_y_mesh()
429            .disable_y_axis()
430            .axis_style(params.colors.axis)
431            .draw()?;
432        let mut stream_chart = ChartBuilder::on(&stream_area)
433            .x_label_area_size(params.area_margin.x)
434            .y_label_area_size(params.area_margin.y)
435            .build_cartesian_2d(x_range.clone(), 0..4)?;
436        stream_chart
437            .configure_mesh()
438            .disable_y_mesh()
439            .disable_y_axis()
440            .axis_style(params.colors.axis)
441            .draw()?;
442
443        for time in [
444            match ds.vantage_point {
445                VantagePoint::Client => request.time_first_headers_tx,
446                VantagePoint::Server => request.time_first_headers_rx,
447            },
448            match ds.vantage_point {
449                VantagePoint::Client => request.time_first_headers_rx,
450                VantagePoint::Server => request.time_first_headers_tx,
451            },
452        ]
453        .into_iter()
454        .flatten()
455        {
456            data_chart.draw_series([TriangleMarker::new((time, 2), 5, color)])?;
457            stream_chart.draw_series([TriangleMarker::new(
458                (time, 2),
459                5,
460                color,
461            )])?;
462        }
463        data_chart.draw_series(points.iter().map(|(time, _)| {
464            PathElement::new([(*time, 2), (*time, 0)], color)
465        }))?;
466        if let Some(stream_frames) = stream_frame_series.get(stream_id) {
467            stream_chart.draw_series(stream_frames.iter().map(|(time, _)| {
468                PathElement::new([(*time, 2), (*time, 0)], color)
469            }))?;
470        }
471
472        let mut length_chart = ChartBuilder::on(&bubble_area)
473            .x_label_area_size(params.area_margin.x)
474            .y_label_area_size(params.area_margin.y)
475            .build_cartesian_2d(x_range.clone(), 0..max_data)?;
476        length_chart
477            .configure_mesh()
478            .disable_y_mesh()
479            .disable_y_axis()
480            .axis_style(params.colors.axis)
481            .draw()?;
482        length_chart.draw_series(points.iter().map(|(time, length)| {
483            PathElement::new([(*time, 0), (*time, *length)], color)
484        }))?;
485    }
486    root.present()?;
487    Ok(())
488}

510    #[test]
511    fn stream_colors_use_union_of_visible_request_and_frame_ids() {
512        let params = test_params();
513        let mut ds = Datastore {
514            application_proto: ApplicationProto::Http3,
515            ..Datastore::default()
516        };
517        // A frame-only stream 1 shifts the positional palette: stream 9
518        // would be color #2 in request lanes but color #3 in frame lanes.
519        // QUIC unidirectional IDs 2 and 6 are excluded by bidi_only.
520        for stream_id in [9, 2, 5] {
521            ds.http_requests
522                .insert(stream_id, HttpRequestStub::default());
523        }
524        let frame_series = BTreeMap::from([
525            (1, vec![(0.0, 2)]),
526            (5, vec![(0.2, 2)]),
527            (6, vec![(0.5, 2)]),
528            (9, vec![(1.0, 2)]),
529        ]);
530
531        let colors = colors_for_visible_streams(&params, &ds, &frame_series);
532        assert_eq!(colors.keys().copied().collect::<Vec<_>>(), vec![1, 5, 9]);
533
534        let mut expected = ColorCycle::default();
535        for stream_id in [1, 5, 9] {
536            assert_eq!(colors[&stream_id], expected.next_color());
537        }
538        let mut former_request_palette = ColorCycle::default();
539        let _first_request_color = former_request_palette.next_color();
540        assert_ne!(colors[&9], former_request_palette.next_color());
541
542        // The mapping is independent of insertion order and remains stable
543        // when we repeat the same selection for all three plot sections.
544        assert_eq!(
545            colors_for_visible_streams(&params, &ds, &frame_series),
546            colors
547        );
548
549        // HTTP/2 has no QUIC direction bits: IDs 2 and 6 become visible.
550        ds.application_proto = ApplicationProto::Http2;
551        let h2 = colors_for_visible_streams(&params, &ds, &frame_series);
552        assert_eq!(h2.keys().copied().collect::<Vec<_>>(), vec![1, 2, 5, 6, 9]);
553    }

613    fn http2_request_ids_not_filtered_and_large_data_lengths_render(
614    ) -> Result<(), Box<dyn Error>> {
615        let params = test_params();
616        let mut ds = Datastore {
617            application_proto: ApplicationProto::Http2,
618            vantage_point: VantagePoint::Server,
619            ..Datastore::default()
620        };
621        let request = HttpRequestStub {
622            time_data_tx_set: vec![(1.0, 9), (4.0, 7), (7.0, 1)],
623            time_first_headers_rx: Some(0.5),
624            ..HttpRequestStub::default()
625        };
626        // HTTP/2 stream ID 3 is not QUIC unidirectional and must survive
627        // bidi_only.
628        ds.http_requests.insert(3, request);
629        let mut series = SeriesStore {
630            sent_x_min: 0.0,
631            sent_x_max: 8.0,
632            ..SeriesStore::default()
633        };
634        series
635            .sent_data_frames_series
636            .insert(3, vec![(1.0, 3), (6.0, 5)]);
637        let out = TempCharts::new()?;
638        plot_stream_multiplexing(&params, "h2-id3", &series, &ds, &out.output())?;
639        let path = out.only_file()?;
640        let rendered = image::open(&path)?.to_rgb8();
641        let layout = MultiplexLayout::new(&params, 1)?;
642        assert_eq!(
643            (rendered.width(), rendered.height()),
644            (layout.width, layout.height)
645        );
646        // Compare to same request without STREAM/DATA samples. Chart geometry
647        // remains identical; pixel differences prove the H2 stream was drawn.
648        let no_frames = SeriesStore {
649            sent_x_min: 0.0,
650            sent_x_max: 8.0,
651            ..SeriesStore::default()
652        };
653        // No frame series, preserving the source time bounds.
654        let out_without = TempCharts::new()?;
655        plot_stream_multiplexing(
656            &params,
657            "h2-id3",
658            &no_frames,
659            &ds,
660            &out_without.output(),
661        )?;
662        let without = image::open(out_without.only_file()?)?.to_rgb8();
663        assert_ne!(
664            rendered.as_raw(),
665            without.as_raw(),
666            "HTTP/2 STREAM lane must render DATA frames"
667        );
668
669        // HTTP/3 does have unidirectional stream IDs; bidi-only filters them.
670        ds.application_proto = ApplicationProto::Http3;
671        let filtered = TempCharts::new()?;
672        plot_stream_multiplexing(
673            &params,
674            "h3-id3",
675            &series,
676            &ds,
677            &filtered.output(),
678        )?;
679        let (_, filtered_height) =
680            image::image_dimensions(filtered.only_file()?)?;
681        assert_eq!(filtered_height, MultiplexLayout::new(&params, 0)?.height);
682        assert!(filtered_height < layout.height);
683        Ok(())
684    }

687    fn multiple_requests_preserve_sorted_lanes_and_vantage_specific_data(
688    ) -> Result<(), Box<dyn Error>> {
689        let params = test_params();
690        let mut datastore = Datastore {
691            application_proto: ApplicationProto::Http2,
692            vantage_point: VantagePoint::Client,
693            ..Datastore::default()
694        };
695        // Insert out of stream-ID order; each request must retain its own
696        // distinct payload and grid row, whichever vantage is selected.
697        datastore.http_requests.insert(7, HttpRequestStub {
698            time_data_rx_set: vec![(3.0, 8), (8.0, 12)],
699            time_data_tx_set: vec![(2.0, 3)],
700            ..HttpRequestStub::default()
701        });
702        datastore.http_requests.insert(3, HttpRequestStub {
703            time_data_rx_set: vec![(1.0, 30)],
704            time_data_tx_set: vec![(6.0, 5), (9.0, 11)],
705            ..HttpRequestStub::default()
706        });
707        let series = SeriesStore {
708            received_x_min: 0.0,
709            received_x_max: 10.0,
710            sent_x_min: 0.0,
711            sent_x_max: 10.0,
712            ..SeriesStore::default()
713        };
714
715        let client_output = TempCharts::new()?;
716        plot_stream_multiplexing(
717            &params,
718            "multi-request",
719            &series,
720            &datastore,
721            &client_output.output(),
722        )?;
723        let client_png = image::open(client_output.only_file()?)?.to_rgb8();
724        let expected_height = MultiplexLayout::new(&params, 2)?.height;
725        assert_eq!(client_png.dimensions(), (params.width, expected_height));
726
727        datastore.vantage_point = VantagePoint::Server;
728        let server_output = TempCharts::new()?;
729        plot_stream_multiplexing(
730            &params,
731            "multi-request",
732            &series,
733            &datastore,
734            &server_output.output(),
735        )?;
736        let server_png = image::open(server_output.only_file()?)?.to_rgb8();
737        assert_eq!(server_png.dimensions(), client_png.dimensions());
738        assert_ne!(
739            client_png.as_raw(),
740            server_png.as_raw(),
741            "client RX and server TX data must produce different lane plots"
742        );
743        Ok(())
744    }
