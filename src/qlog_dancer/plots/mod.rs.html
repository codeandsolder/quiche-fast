71/// Encode user-controlled names as single path components without losing
72/// information or creating collisions between escaped and literal strings.
73#[cfg(not(target_arch = "wasm32"))]
74fn chart_filename_component(value: &str) -> String {
75    const HEX: &[u8; 16] = b"0123456789ABCDEF";
76    let mut safe = String::with_capacity(value.len());
77    for byte in value.bytes() {
78        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'_' | b'.') {
79            safe.push(char::from(byte));
80        } else {
81            safe.push('~');
82            safe.push(char::from(HEX[usize::from(byte >> 4)]));
83            safe.push(char::from(HEX[usize::from(byte & 0x0F)]));
84        }
85    }
86    safe
87}

89impl ChartConfig {
90    /// Keep PNG filename metadata native-only. Browser plots need only their
91    /// canvas output type, and constructing the configuration is identical
92    /// across all plotters.
93    fn new(
94        title: &str, filename: &str, clamp: &ClampParams, ds: &Datastore,
95        ty: &ChartOutputType,
96    ) -> Self {
97        // Canvas only needs its output ID. These inputs are used for native
98        // PNG filenames, so keep one consistent constructor on both targets.
99        #[cfg(target_arch = "wasm32")]
100        let _ = (title, filename, clamp, ds);
101
102        Self {
103            #[cfg(not(target_arch = "wasm32"))]
104            title: title.to_owned(),
105            #[cfg(not(target_arch = "wasm32"))]
106            input_filename: filename.to_owned(),
107            #[cfg(not(target_arch = "wasm32"))]
108            clamp: clamp.clone(),
109            #[cfg(not(target_arch = "wasm32"))]
110            app_proto: ds.application_proto,
111            #[cfg(not(target_arch = "wasm32"))]
112            host: ds.host.clone(),
113            #[cfg(not(target_arch = "wasm32"))]
114            session_id: ds.session_id,
115            ty: ty.clone(),
116        }
117    }
118
119    #[cfg(not(target_arch = "wasm32"))]
120    pub fn chart_name(&self) -> String {
121        let proto = match self.app_proto {
122            ApplicationProto::Http2 => "h2",
123            ApplicationProto::Http3 => "h3",
124        };
125
126        let host = match (&self.host, &self.ty) {
127            (Some(host), ChartOutputType::Png { .. }) => {
128                // Keep the established separator for native PNG files.
129                let encoded = chart_filename_component(host);
130                format!("-{encoded}")
131            },
132            (Some(host), ChartOutputType::Canvas { .. }) =>
133                chart_filename_component(host),
134            _ => String::new(),
135        };
136        let session = self
137            .session_id
138            .map_or_else(String::new, |id| format!("-session{id}"));
139        let start = self
140            .clamp
141            .start
142            .map_or_else(String::new, |value| format!("-s{value}"));
143        let end = self
144            .clamp
145            .end
146            .map_or_else(String::new, |value| format!("-e{value}"));
147
148        let input_filename = chart_filename_component(&self.input_filename);
149        let title = chart_filename_component(&self.title);
150        let mut name = format!(
151            "{input_filename}-{proto}{host}{session}-{title}{start}{end}"
152        );
153
154        if let ChartOutputType::Png {
155            cwnd_y_max,
156            stream_y_max,
157            ..
158        } = &self.ty
159        {
160            if let Some(maximum) = stream_y_max {
161                let _ = write!(&mut name, "-stream_max_y{maximum}");
162            }
163            if let Some(maximum) = cwnd_y_max {
164                let _ = write!(&mut name, "-cwnd_max_y{maximum}");
165            }
166        }
167
168        name
169    }
170
171    #[cfg(not(target_arch = "wasm32"))]
172    pub fn chart_filepath(&self) -> String {
173        match &self.ty {
174            ChartOutputType::Png { output_dir, .. } => {
175                let name = self.chart_name();
176                format!("{output_dir}/{name}.png")
177            },
178
179            // Canvas doesn't have a file
180            ChartOutputType::Canvas { .. } => self.chart_name(),
181        }
182    }
183
184    #[cfg(target_arch = "wasm32")]
185    fn canvas_id(&self) -> Option<String> {
186        match &self.ty {
187            ChartOutputType::Canvas { canvas_id } => Some(canvas_id.clone()),
188            ChartOutputType::Png { .. } => None,
189        }
190    }
191}

249fn draw_line<DB: DrawingBackend>(
250    data: &[(f64, u64)], label: Option<&str>, colour: RGBColor,
251    chart: &mut ChartContext<DB, Cartesian2d<RangedCoordf64, RangedCoordu64>>,
252) -> Result<(), plotters::drawing::DrawingAreaErrorKind<DB::ErrorType>> {
253    let drawn =
254        chart.draw_series(LineSeries::new(data.iter().copied(), colour))?;
255    if let Some(label) = label {
256        drawn.label(label).legend(move |(x, y)| {
257            PathElement::new([(x, y), (x + 20, y)], colour)
258        });
259    }
260    Ok(())
261}

284    #[test]
285    fn chart_names_preserve_metadata_across_png_and_canvas() {
286        let ds = Datastore {
287            application_proto: ApplicationProto::Http2,
288            host: Some("api.example:443".into()),
289            session_id: Some(42),
290            ..Datastore::default()
291        };
292        let clamp = ClampParams {
293            start: Some(1.5),
294            end: Some(3.25),
295            stream_y_max: None,
296        };
297        let png = ChartOutputType::Png {
298            output_dir: "/tmp/charts".into(),
299            cwnd_y_max: Some(80),
300            stream_y_max: Some(40),
301        };
302        let chart =
303            ChartConfig::new("overview", "logs/input.qlog", &clamp, &ds, &png);
304        let expected = "logs~2Finput.qlog-h2-api.example~3A443-session42-overview-s1.5-e3.25-stream_max_y40-cwnd_max_y80";
305        assert_eq!(chart.chart_name(), expected);
306        assert_eq!(
307            chart.chart_filepath(),
308            format!("/tmp/charts/{expected}.png")
309        );
310
311        let canvas = ChartOutputType::Canvas {
312            canvas_id: "main".into(),
313        };
314        let chart =
315            ChartConfig::new("overview", "logs/input.qlog", &clamp, &ds, &canvas);
316        assert_eq!(
317            chart.chart_name(),
318            "logs~2Finput.qlog-h2api.example~3A443-session42-overview-s1.5-e3.25"
319        );
320        assert_eq!(chart.chart_filepath(), chart.chart_name());
321
322        let default_ds = Datastore::default();
323        let chart = ChartConfig::new(
324            "overview",
325            "input.qlog",
326            &ClampParams::default(),
327            &default_ds,
328            &canvas,
329        );
330        assert_eq!(chart.chart_name(), "input.qlog-h3-overview");
331    }

333    #[test]
334    fn chart_filenames_escape_untrusted_components() -> Result<(), Box<dyn Error>>
335    {
336        assert_eq!(chart_filename_component("a/b"), "a~2Fb");
337        assert_eq!(chart_filename_component("a~2Fb"), "a~7E2Fb");
338        assert_eq!(chart_filename_component(r"../\evil:"), "..~2F~5Cevil~3A");
339        assert_eq!(chart_filename_component("é"), "~C3~A9");
340        let ds = Datastore {
341            host: Some("../../outside:443/".into()),
342            ..Datastore::default()
343        };
344        let output_dir = "/tmp/qlog-test-chart-destination";
345        let ty = ChartOutputType::Png {
346            output_dir: output_dir.into(),
347            cwnd_y_max: None,
348            stream_y_max: None,
349        };
350        let config = ChartConfig::new(
351            "conn-overview",
352            "../sample.qlog",
353            &ClampParams::default(),
354            &ds,
355            &ty,
356        );
357        let path = config.chart_filepath();
358        assert_eq!(Path::new(&path).parent(), Some(Path::new(output_dir)));
359        let component = Path::new(&path)
360            .file_name()
361            .ok_or("missing chart filename")?
362            .to_string_lossy();
363        assert!(component.contains("~2F"));
364        assert!(!component.contains('/'));
365        assert!(!component.contains('\\'));
366        Ok(())
367    }
