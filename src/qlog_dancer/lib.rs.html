49#[derive(PartialEq, Eq, PartialOrd, Ord, Hash, Debug, Copy, Clone)]
50pub enum PacketType {
51    Initial,
52    Handshake,
53    ZeroRtt,
54    OneRtt,
55    Retry,
56    VersionNegotiation,
57    Unknown,
58}
59
60impl PacketType {
61    /// Classify a typed qlog packet, mapping unknown and stateless-reset
62    /// packets to the generic bucket used by the plots.
63    #[must_use = "the returned value identifies the converted packet type"]
64    pub const fn from_qlog_packet_type(
65        ty: &qlog::events::quic::PacketType,
66    ) -> Self {
67        use qlog::events::quic::PacketType as QlogPacketType;
68
69        match ty {
70            QlogPacketType::Initial => Self::Initial,
71            QlogPacketType::Handshake => Self::Handshake,
72            QlogPacketType::ZeroRtt => Self::ZeroRtt,
73            QlogPacketType::OneRtt => Self::OneRtt,
74            QlogPacketType::Retry => Self::Retry,
75            QlogPacketType::VersionNegotiation => Self::VersionNegotiation,
76            QlogPacketType::StatelessReset | QlogPacketType::Unknown =>
77                Self::Unknown,
78        }
79    }
80
81    /// Classify a packet from Chrome netlog header metadata.
82    #[must_use = "the returned value identifies the decoded packet type"]
83    pub fn from_netlog_packet_header(
84        header_format: &str, long_header_type: Option<&str>,
85    ) -> Self {
86        match header_format {
87            "IETF_QUIC_LONG_HEADER_PACKET" => match long_header_type {
88                Some("INITIAL") => Self::Initial,
89                Some("HANDSHAKE") => Self::Handshake,
90                _ => Self::Unknown,
91            },
92            "IETF_QUIC_SHORT_HEADER_PACKET" => Self::OneRtt,
93            _ => Self::Unknown,
94        }
95    }
96
97    /// Classify a packet from Chrome netlog's encryption level.
98    #[must_use = "the returned value identifies the packet encryption level"]
99    pub fn from_netlog_encryption_level(encryption_level: &str) -> Self {
100        match encryption_level {
101            "ENCRYPTION_INITIAL" => Self::Initial,
102            "ENCRYPTION_HANDSHAKE" => Self::Handshake,
103            "ENCRYPTION_ZERO_RTT" => Self::ZeroRtt,
104            "ENCRYPTION_FORWARD_SECURE" => Self::OneRtt,
105            _ => Self::Unknown,
106        }
107    }
108}

110#[derive(Debug, Clone, Copy, PartialEq, Eq)]
111pub enum SerializationFormat {
112    QlogJson,
113    QlogJsonSeq,
114    NetlogJson,
115    Unknown,
116}
117
118impl SerializationFormat {
119    #[must_use = "the returned format determines how to parse the input file"]
120    pub fn from_file_extension(extension: &str) -> Self {
121        match extension {
122            ext if ext.eq_ignore_ascii_case("qlog") => Self::QlogJson,
123            ext if ["sqlog", "gz", "zst"]
124                .iter()
125                .any(|suffix| ext.eq_ignore_ascii_case(suffix)) =>
126                Self::QlogJsonSeq,
127            ext if ext.eq_ignore_ascii_case("json") => Self::NetlogJson,
128            _ => Self::Unknown,
129        }
130    }
131
132    #[must_use = "the returned format determines how to parse the input file"]
133    pub fn from_filename(filename: &str) -> Self {
134        // Path handles the final suffix of compound extensions like .sqlog.gz.
135        std::path::Path::new(filename)
136            .extension()
137            .and_then(std::ffi::OsStr::to_str)
138            .map_or(Self::Unknown, Self::from_file_extension)
139    }
140}

142/// Identity of one independent qlog trace. Indices are zero-based and
143/// correspond to the trace's position in the source document.
144#[derive(Clone, Debug, PartialEq, Eq)]
145pub struct QlogTraceIdentity {
146    pub index: usize,
147    pub title: Option<String>,
148    pub vantage_point: qlog::VantagePointType,
149}
150
151impl QlogTraceIdentity {
152    #[must_use = "the returned string contains the trace label"]
153    pub fn label(&self) -> String {
154        let prefix = format!("trace {}", self.index + 1);
155        self.title
156            .as_deref()
157            .map_or_else(|| prefix.clone(), |title| format!("{prefix}: {title}"))
158    }
159}

192/// Parse the configured qlog JSON, JSON-SEQ or Chrome netlog input.
193///
194/// # Errors
195/// Returns an error for an unknown format, malformed input, unsupported
196/// vantage points, invalid filters or datastore processing failures.
197pub fn parse_log_file(
198    config: &AppConfig,
199) -> Result<LogFileParseResult, Box<dyn Error>> {
200    match config.log_format {
201        SerializationFormat::QlogJson => parse_qlog_json(config),
202        SerializationFormat::QlogJsonSeq => parse_qlog_json_seq(config),
203        SerializationFormat::NetlogJson => parse_netlog_json(config),
204        SerializationFormat::Unknown => {
205            error!("Unknown log file format for {}", config.filename);
206            Err(std::io::Error::new(
207                std::io::ErrorKind::InvalidInput,
208                format!("unknown log file format for {}", config.filename),
209            )
210            .into())
211        },
212    }
213}

215fn parse_qlog_json(
216    config: &AppConfig,
217) -> Result<LogFileParseResult, Box<dyn Error>> {
218    let started = std::time::Instant::now();
219    let qlog = read_qlog_from_file(&config.file)?;
220    if qlog.traces.is_empty() {
221        return Err(std::io::Error::new(
222            std::io::ErrorKind::InvalidData,
223            "qlog file contains no traces",
224        )
225        .into());
226    }
227    debug!("parsed qlog JSON in {:?}", started.elapsed());
228
229    let started = std::time::Instant::now();
230    let mut data = Vec::with_capacity(qlog.traces.len());
231    for (index, trace) in qlog.traces.into_iter().enumerate() {
232        let vantage_point = trace
233            .vantage_point
234            .as_ref()
235            .map_or(qlog::VantagePointType::Unknown, |vp| vp.ty.clone());
236        let datastore = Datastore::with_qlog_events(
237            &trace.events,
238            &vantage_point,
239            !config.ignore_acks,
240        )?;
241        data.push(LogFileData {
242            datastore,
243            raw: RawLogEvents::QlogJson {
244                events: trace.events,
245            },
246            trace: Some(QlogTraceIdentity {
247                index,
248                title: trace.title,
249                vantage_point,
250            }),
251        });
252    }
253    debug!(
254        "populated {} independent qlog datastores in {:?}",
255        data.len(),
256        started.elapsed()
257    );
258    Ok(LogFileParseResult {
259        details: LogFileDetails {
260            file_schema: qlog.file_schema,
261            serialization_format: qlog.serialization_format,
262            sessions: BTreeMap::new(),
263        },
264        data,
265    })
266}

268fn parse_qlog_json_seq(
269    config: &AppConfig,
270) -> Result<LogFileParseResult, Box<dyn Error>> {
271    println!("parsing qlog as JSON-SEQ...");
272    let started = std::time::Instant::now();
273    let (qlog_reader, details) = qlog_seq_reader(config)?;
274    debug!("	complete in {:?}", started.elapsed());
275
276    println!("populating datastore...");
277    let started = std::time::Instant::now();
278    let trace = QlogTraceIdentity {
279        index: 0,
280        title: qlog_reader.qlog.trace.title.clone(),
281        vantage_point: qlog_reader
282            .qlog
283            .trace
284            .vantage_point
285            .as_ref()
286            .map_or(qlog::VantagePointType::Unknown, |vp| vp.ty.clone()),
287    };
288    let events: Vec<qlog::reader::Event> = qlog_reader.into_iter().collect();
289    let events = if let Some(filter) = &config.qlog_wirefilter {
290        filter_sqlog_events(events, filter)?
291    } else {
292        events
293    };
294    // Keep plot series, stats, and raw event exports in agreement.
295    let datastore = Datastore::with_sqlog_reader_events(
296        &events,
297        &trace.vantage_point,
298        !config.ignore_acks,
299    )?;
300    debug!("	complete in {:?}", started.elapsed());
301
302    Ok(LogFileParseResult {
303        details,
304        data: vec![LogFileData {
305            datastore,
306            raw: RawLogEvents::QlogJsonSeq { events },
307            trace: Some(trace),
308        }],
309    })
310}

312fn parse_netlog_json(
313    config: &AppConfig,
314) -> Result<LogFileParseResult, Box<dyn Error>> {
315    println!("setting up parsing file as netlog...");
316    let started = std::time::Instant::now();
317    let file = File::open(&config.file)?;
318    let mut reader = BufReader::new(file);
319    let constants = netlog::read_netlog_constants(&mut reader)?;
320    debug!("	complete in {:?}", started.elapsed());
321
322    println!("parsing file ...");
323    println!("Filtering on hostnames: {:#?}", config.netlog_filter);
324    let started = std::time::Instant::now();
325    let (data, sessions) = datastore::with_netlog_reader(
326        &mut reader,
327        &config.netlog_filter,
328        &constants,
329    )?;
330    debug!("	complete in {:?}", started.elapsed());
331
332    Ok(LogFileParseResult {
333        details: LogFileDetails {
334            file_schema: constants.log_format_version.to_string(),
335            serialization_format: "Chrome netlog".to_string(),
336            sessions,
337        },
338        data,
339    })
340}

342/// Read a qlog JSON document from disk.
343///
344/// # Errors
345/// Returns an error if the file cannot be opened or decoded as qlog JSON.
346pub fn read_qlog_from_file<P: AsRef<Path>>(
347    path: P,
348) -> Result<Qlog, Box<dyn Error>> {
349    let file = File::open(path)?;
350    let reader = BufReader::new(file);
351
352    let qlog = serde_json::from_reader(reader)?;
353
354    Ok(qlog)
355}

357/// Open a sequential qlog file, including supported compressed formats.
358///
359/// # Errors
360/// Returns an error for an unsupported file extension, invalid header,
361/// unavailable decompressor, or an I/O failure.
362pub fn qlog_seq_reader(
363    config: &AppConfig,
364) -> Result<(QlogSeqReader<'static>, LogFileDetails), Box<dyn Error>> {
365    let qlog_reader =
366        QlogSeqReader::with_file(config.file.clone()).map_err(|e| {
367            std::io::Error::other(format!("problem reading qlog file: {e}"))
368        })?;
369
370    let log_file_details = LogFileDetails {
371        file_schema: qlog_reader.qlog.file_schema.clone(),
372        serialization_format: qlog_reader.qlog.serialization_format.clone(),
373        sessions: BTreeMap::new(),
374    };
375
376    Ok((qlog_reader, log_file_details))
377}

379pub fn stringify_last<T>(src: &[T]) -> String
380where
381    T: std::fmt::Debug,
382{
383    match src {
384        [] | [_] => "n/a".to_string(),
385        [.., last] => format!("{last:?}"),
386    }
387}

389// slight hack: duplicate the previous point so
390// that no misleading line interpolation occurs
391fn push_interp<X: Clone, Y: Clone>(collection: &mut Vec<(X, Y)>, value: (X, Y)) {
392    let prev = collection.last().cloned();
393    let new_time = value.0.clone();
394
395    if let Some((_, y)) = prev {
396        collection.push((new_time, y));
397    }
398
399    collection.push(value);
400}

402fn create_file_recursive(filename: &str) -> std::io::Result<File> {
403    let path = std::path::Path::new(filename);
404    if let Some(dir) = path.parent() {
405        std::fs::create_dir_all(dir)?;
406    }
407
408    File::create(filename)
409}

411/// Split the category at the first separator, preserving arbitrary colons
412/// within custom event names.
413#[must_use = "the returned pair contains the event category and type"]
414pub fn category_and_type_from_name(name: &str) -> (String, String) {
415    match name.split_once(':') {
416        Some((category, event_name)) =>
417            (category.to_owned(), event_name.to_owned()),
418        None => (name.to_owned(), String::new()),
419    }
420}
421
422/// Derive the category and full event name from a serializable log record.
423///
424/// # Errors
425/// Returns the serialization error (or a descriptive missing-name error)
426/// rather than panicking when an event cannot be serialized.
427pub fn category_and_type_from_event<T: Serialize>(
428    ev: &T,
429) -> Result<(String, String), serde_json::Error> {
430    let encoded = serde_json::to_value(ev)?;
431    let name = encoded
432        .get("name")
433        .and_then(serde_json::Value::as_str)
434        .ok_or_else(|| {
435            serde_json::Error::io(std::io::Error::new(
436                std::io::ErrorKind::InvalidData,
437                "serialized qlog event has no string name",
438            ))
439        })?;
440    Ok(category_and_type_from_name(name))
441}

690                &self, _serializer: S,
691            ) -> Result<S::Ok, S::Error> {
692                Err(serde::ser::Error::custom("event serializer failure"))
693            }
694        }
695        assert!(
696            super::category_and_type_from_event(&BrokenEvent).is_err_and(
697                |error| error.to_string().contains("event serializer failure")
698            )
699        );
700        assert_eq!(
701            super::category_and_type_from_name("quic:some:future:event"),
702            ("quic".to_owned(), "some:future:event".to_owned()),
703        );
704        assert_eq!(
705            super::category_and_type_from_event(&serde_json::json!({
706                "name": "custom:event:type", "data": {"x": 1}
707            }))
708            .ok(),
709            Some(("custom".to_owned(), "event:type".to_owned())),
710        );
711        assert!(super::category_and_type_from_event(
712            &serde_json::json!({"x": 1})
713        )
714        .is_err());
715    }
